Powiadomienia (Toasts)

Toasty stackują się automatycznie i posiadają animowany pasek postępu. 

<!-- Wymaga pliku molique-script.js -->
<script>
  MoliqueToast.show({
    message: 'Zapisano pomyślnie!',
    type: 'success',
    position: 'top-right',
    duration: 4000
  });
</script>
