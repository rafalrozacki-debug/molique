Toasts stapeln sich automatisch und haben einen animierten Fortschrittsbalken. 

<!-- Erfordert die Datei molique-script.js -->
<script>
  MoliqueToast.show({
    message: 'Erfolgreich gespeichert!',
    type: 'success',
    position: 'top-right',
    duration: 4000
  });
</script>
